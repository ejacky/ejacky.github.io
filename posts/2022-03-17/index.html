    for-select 结构里的 break 关键字

    描述信息
    March 17, 2022 · 作者

    for-select#

    package test

import (
	"fmt"
	"testing"
	"time"
)

func TestBreak(t *testing.T) {
	tick := time.Tick(time.Second)
	for {
		select {
		case t := <-tick:
			fmt.Println(t)
			break
		}
	}
	fmt.Println("end")
}
